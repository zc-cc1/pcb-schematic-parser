cmake_minimum_required(VERSION 3.20)
project(opencv_week01 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()

# 默认只学习纯C++。预设也显式关闭此开关，避免旧缓存ON继续查找OpenCV。
option(WEEK01_WITH_OPENCV "Build image demos requiring OpenCV C++ SDK" OFF)
add_executable(basics basics.cpp)
if(MSVC)
    target_compile_options(basics PRIVATE /utf-8 /W4)
endif()
enable_testing()
add_test(NAME cpp_basics COMMAND basics)

if(WEEK01_WITH_OPENCV)
    message(STATUS "OpenCV examples enabled: looking for OpenCV C++ SDK")
    find_package(OpenCV 4 REQUIRED COMPONENTS core imgcodecs)
    foreach(name demo exercise_solution)
        add_executable(${name} ${name}.cpp)
        target_include_directories(${name} PRIVATE ${OpenCV_INCLUDE_DIRS})
        target_link_libraries(${name} PRIVATE ${OpenCV_LIBS})
        if(MSVC)
            target_compile_options(${name} PRIVATE /utf-8 /W4)
        endif()
        # 输出位置随CMake配置为源码目录/outputs，不依赖启动时工作目录。
        target_compile_definitions(${name} PRIVATE WEEK01_OUTPUT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/outputs")
        add_test(NAME ${name}_checks COMMAND ${name})
    endforeach()
else()
    message(STATUS "Basics-only mode: OpenCV is not required. Building target: basics")
endif()
